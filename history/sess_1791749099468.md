def main(stdscr):

    curses.curs_set(0)

    height, width = stdscr.getmaxyx(), stdscr.getmaxyx()
    board = [[' ' for _ in range(width)] for _ in range(height)]

    def draw():
        stdscr.clear()
        for i in range(height):
            for j in range(width):
                if board[i][j] != ' ':
                    ch = board[i][j]
                    if ch == ' ':
                        ch = '.'
                    stdscr.addch(i, j, ch)
            stdscr.refresh()
        stdscr.timeout = 1000
        stdscr.waitkeypad()

    def check_win():
        for i in range(height):
            for j in range(width):
                if board[i][j] != ' ' and board[i][j] == board[i+1][j] == board[i+2][j] == 'X':
                    return 'X'
                if board[i][j] != ' ' and board[i][j] == board[i][j+1] == board[i][j+2] == 'X':
                    return 'X'
                if board[i][j] != ' ' and board[i][j] == board[i][j+1] == board[i][j+2] == 'O':
                    return 'O'
                if board[i][j] != ' ' and board[i][j] == board[i+1][j] == board[i+2][j] == 'O':
                    return 'O'
                if board[i][j] != ' ' and board[i][j] == board[i][j+1] == board[i][j+2] == 'X':
                    return 'X'
                if board[i][j] != ' ' and board[i][j] == board[i+1][j] == board[i+2][j] == 'O':
                    return 'O'
        return None

    def move():
        key = stdscr.getch()
        if key == ord('q'):
            return
        col = key % width
        row = key // width
        if board[row][col] == ' ':
            board[row][col] = 'X' if turn == 0 else 'O'
            turn = 1 - turn
        else:
            print('Invalid move')
        draw()

        move()
        time.sleep(0.3)

if __name__ == '__main__':
    curses.wrapper(main)
